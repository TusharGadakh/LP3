Design n-Queens matrix having first Queen placed. Use backtracking to place remaining Queens to generate the final n-queen‘s matrix.


#include <iostream>
using namespace std;

int board[20][20], n;

// Check whether Queen can be placed safely
bool isSafe(int row, int col) {

    // Check column
    for (int i = 0; i < row; i++) {
        if (board[i][col] == 1)
            return false;
    }

    // Check left diagonal
    for (int i = row - 1, j = col - 1;
         i >= 0 && j >= 0; i--, j--) {
        if (board[i][j] == 1)
            return false;
    }

    // Check right diagonal
    for (int i = row - 1, j = col + 1;
         i >= 0 && j < n; i--, j++) {
        if (board[i][j] == 1)
            return false;
    }

    return true;
}

// Solve using backtracking
bool solve(int row) {

    if (row == n)
        return true;

    for (int col = 0; col < n; col++) {

        if (isSafe(row, col)) {

            board[row][col] = 1;

            if (solve(row + 1))
                return true;

            // Backtracking
            board[row][col] = 0;
        }
    }

    return false;
}

int main() {

    int firstCol;

    cout << "Enter number of Queens: ";
    cin >> n;

    if (n < 1 || n > 20) {
        cout << "Enter n between 1 and 20.";
        return 0;
    }

    cout << "Enter first Queen column (1 to " << n << "): ";
    cin >> firstCol;

    if (firstCol < 1 || firstCol > n) {
        cout << "Invalid column.";
        return 0;
    }

    // Place first Queen
    board[0][firstCol - 1] = 1;

    if (solve(1)) {

        cout << "\nFinal N-Queens Matrix:\n";

        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                cout << board[i][j] << " ";
            }
            cout << endl;
        }

    }
    else {
        cout << "Solution does not exist.";
    }

    return 0;
}
